cmake_minimum_required(VERSION 3.22)
project(gyroflow_ocio_native_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.7.3 REQUIRED COMPONENTS Core Gui Quick Qml ShaderTools)
if(NOT TARGET Qt6::GuiPrivate)
    find_package(Qt6GuiPrivate CONFIG REQUIRED)
endif()
if(NOT TARGET Qt6::ShaderToolsPrivate)
    find_package(Qt6ShaderToolsPrivate CONFIG REQUIRED)
endif()
find_package(OpenColorIO 2.4.2 EXACT CONFIG REQUIRED)
add_executable(native-preview native_preview.cpp ../../src/rendering/ocio/bridge.cpp)
target_link_libraries(native-preview PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate
    Qt6::Quick Qt6::Qml Qt6::ShaderTools Qt6::ShaderToolsPrivate OpenColorIO::OpenColorIO)
set(OCIO_PREVIEW_LUT_FILE "" CACHE FILEPATH "Optional locally owned official O4 .cube for native pixel acceptance")
enable_testing()
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_test(NAME native-float32-preview COMMAND native-preview "${REPOSITORY_ROOT}"
    "${OCIO_PREVIEW_LUT_FILE}" "${CMAKE_CURRENT_BINARY_DIR}/results.json")
add_test(NAME software-renderer-rejection COMMAND native-preview --software-check)
add_test(NAME shader-exception-lock-recovery COMMAND native-preview --exception-check)
set_tests_properties(shader-exception-lock-recovery PROPERTIES TIMEOUT 30)
